<script>
    // -----------------------------------------------------
    // ステージデータ定義（画像を排除し、確実な絵文字アイコンに）
    // -----------------------------------------------------
    const stages = {
        classic: {
            maxMoves: 7, // 最短7手
            turnLimit: false,
            entities: [
                { id: 'f1', type: 'farmer', name: '人間', icon: '👦', canDrive: true },
                { id: 'w1', type: 'wolf', name: 'オオカミ', icon: '🐺', canDrive: false },
                { id: 'g1', type: 'goat', name: 'ヤギ', icon: '🐐', canDrive: false },
                { id: 'c1', type: 'cabbage', name: 'キャベツ', icon: '🥬', canDrive: false }
            ]
        },
        twins: {
            maxMoves: 9, // 最短9手
            turnLimit: false,
            entities: [
                { id: 'f1', type: 'farmer', name: '人間A', icon: '👦', canDrive: true },
                { id: 'f2', type: 'farmer', name: '人間B', icon: '👱‍♂️', canDrive: true },
                { id: 'g1', type: 'goat', name: 'ヤギ1', icon: '🐐', canDrive: false },
                { id: 'g2', type: 'goat', name: 'ヤギ2', icon: '🐐', canDrive: false },
                { id: 'c1', type: 'cabbage', name: 'キャベツ1', icon: '🥬', canDrive: false },
                { id: 'c2', type: 'cabbage', name: 'キャベツ2', icon: '🥬', canDrive: false }
            ]
        },
        werewolf: {
            maxMoves: 7, // 最短7手
            turnLimit: true,
            nightTurns: [4, 5],
            entities: [
                { id: 'f1', type: 'farmer', name: '人間A', icon: '👦', canDrive: true },
                { id: 'f2', type: 'werewolf_human', name: '人間B', icon: '👱‍♂️', canDrive: true },
                { id: 'g1', type: 'goat', name: 'ヤギ', icon: '🐐', canDrive: false },
                { id: 'c1', type: 'cabbage', name: 'キャベツ', icon: '🥬', canDrive: false }
            ]
        }
    };

    // -----------------------------------------------------
    // 状態管理
    // -----------------------------------------------------
    let currentStage = 'classic';
    let entities = [];
    let state = []; // 履歴用 (UNDO)
    let currentState = {};
    let isGameOver = false;

    const leftBankEl = document.getElementById('left-bank');
    const rightBankEl = document.getElementById('right-bank');
    const boatEl = document.getElementById('boat');
    const boatSlotsEl = [document.getElementById('slot-0'), document.getElementById('slot-1')];
    const messageEl = document.getElementById('message-area');
    const statusBarEl = document.getElementById('status-bar');
    const durabilityEl = document.getElementById('durability-value');

    // -----------------------------------------------------
    // 初期化と描画
    // -----------------------------------------------------
    function initGame() {
        const stageData = stages[currentStage];
        entities = JSON.parse(JSON.stringify(stageData.entities));
        
        currentState = {
            turn: 1,
            durability: stageData.maxMoves,
            boatPosition: 'left',
            positions: {}
        };
        entities.forEach(e => { currentState.positions[e.id] = 'left'; });
        state = [JSON.parse(JSON.stringify(currentState))];
        isGameOver = false;
        
        boatEl.className = ''; 
        document.body.className = '';
        messageEl.innerHTML = 'アイコンをタップして船に乗せよう。<br>最短手数でないとイカダが壊れます！';
        messageEl.className = '';
        
        render();
    }

    function render() {
        leftBankEl.innerHTML = '';
        rightBankEl.innerHTML = '';
        boatSlotsEl[0].innerHTML = '';
        boatSlotsEl[1].innerHTML = '';

        durabilityEl.innerText = currentState.durability;

        let isNight = stages[currentStage].turnLimit && stages[currentStage].nightTurns.includes(currentState.turn);
        if (isNight) {
            document.body.classList.add('night-mode');
            statusBarEl.innerHTML = `ターン: ${currentState.turn} | 夜 🌕 (人間Bが人狼化!)`;
        } else {
            document.body.classList.remove('night-mode');
            statusBarEl.innerHTML = `ターン: ${currentState.turn} | 昼 ☀️`;
        }

        // キャラクターアイコン描画（画像ではなくCSS+絵文字で確実に描画）
        entities.forEach(entity => {
            const el = document.createElement('div');
            el.className = 'character';
            el.id = entity.id;
            
            let displayIcon = entity.icon;
            let displayName = entity.name;
            if (entity.type === 'werewolf_human' && isNight) {
                displayIcon = '🐺';
                displayName = '人狼';
            }
            
            el.innerHTML = `
                ${displayIcon}
                <div class="character-name-label">${displayName}</div>
            `;
            
            if (!isGameOver) {
                el.onclick = (e) => { e.stopPropagation(); handleCharacterClick(entity.id); };
            }

            const pos = currentState.positions[entity.id];
            if (pos === 'left') leftBankEl.appendChild(el);
            else if (pos === 'right') rightBankEl.appendChild(el);
            else if (pos === 'boat_0') boatSlotsEl[0].appendChild(el);
            else if (pos === 'boat_1') boatSlotsEl[1].appendChild(el);
        });

        if (currentState.boatPosition === 'right') boatEl.classList.add('right-side');
        else boatEl.classList.remove('right-side');

        if (isGameOver && currentState.durability <= 0) boatEl.classList.add('broken');
    }

    // -----------------------------------------------------
    // 操作ロジック
    // -----------------------------------------------------
    function handleCharacterClick(id) {
        if (isGameOver) return;
        const pos = currentState.positions[id];
        
        if (pos.startsWith('boat')) {
            currentState.positions[id] = currentState.boatPosition;
        } else if (pos === currentState.boatPosition) {
            if (!currentState.positions['boat_0'] && Object.values(currentState.positions).indexOf('boat_0') === -1) {
                currentState.positions[id] = 'boat_0';
            } else if (!currentState.positions['boat_1'] && Object.values(currentState.positions).indexOf('boat_1') === -1) {
                currentState.positions[id] = 'boat_1';
            }
        }
        render();
    }

    boatEl.onclick = () => {
        if (isGameOver) return;

        const onBoat = entities.filter(e => currentState.positions[e.id].startsWith('boat'));
        const hasDriver = onBoat.some(e => e.canDrive);
        
        let isWaiting = false;
        if (onBoat.length === 0 && currentStage === 'werewolf') {
            isWaiting = true; // 人狼ステージで誰も乗っていなければ待機
        } else if (!hasDriver) {
            showMessage("船を動かすには人間が必要です！", "game-over");
            return;
        }

        saveState();

        if (!isWaiting) currentState.boatPosition = (currentState.boatPosition === 'left') ? 'right' : 'left';
        
        currentState.turn++;
        currentState.durability--; // 動く（または待機する）たびに耐久値減少
        
        render();
        setTimeout(() => { checkRules(); }, 300);
    };

    function saveState() { state.push(JSON.parse(JSON.stringify(currentState))); }

    function undo() {
        if (state.length > 1) {
            currentState = state.pop();
            isGameOver = false;
            boatEl.classList.remove('broken');
            messageEl.innerHTML = "一手戻しました。";
            messageEl.className = "";
            render();
        }
    }

    // -----------------------------------------------------
    // 勝敗判定
    // -----------------------------------------------------
    function checkRules() {
        const allRight = entities.every(e => currentState.positions[e.id] === 'right');
        if (allRight) {
            showMessage("🎉 全員無事に渡りきりました！クリア！", "success");
            isGameOver = true;
            return;
        }

        const checkBank = (bankName) => {
            const inBank = entities.filter(e => {
                const p = currentState.positions[e.id];
                return p === bankName || (currentState.boatPosition === bankName && p.startsWith('boat'));
            });
            
            const types = inBank.map(e => {
                if (currentStage === 'werewolf' && e.type === 'werewolf_human') {
                    return stages['werewolf'].nightTurns.includes(currentState.turn) ? 'wolf' : 'farmer';
                }
                return e.type;
            });

            if (!types.includes('farmer')) {
                if (types.includes('wolf') && types.includes('goat')) return "オオカミがヤギを食べた！🐺 ＞ 🐐";
                if (types.includes('goat') && types.includes('cabbage')) return "ヤギがキャベツを食べた！🐐 ＞ 🥬";
            }
            return null;
        };

        const leftError = checkBank('left');
        const rightError = checkBank('right');

        if (leftError || rightError) {
            showMessage(`💀 ゲームオーバー！<br>${leftError || rightError}`, "game-over");
            isGameOver = true;
        } else if (currentState.durability <= 0) {
            showMessage("💥 イカダの耐久値が尽きて壊れてしまった！ゲームオーバー！", "game-over");
            isGameOver = true;
            boatEl.classList.add('broken');
        } else {
            messageEl.innerHTML = "無事に移動しました。（残り耐久値：" + currentState.durability + "）";
            messageEl.className = "";
        }
    }

    function showMessage(msg, className) {
        messageEl.innerHTML = msg;
        messageEl.className = className;
    }

    // -----------------------------------------------------
    // イベントリスナー
    // -----------------------------------------------------
    document.getElementById('stage-select').addEventListener('change', (e) => {
        currentStage = e.target.value;
        initGame();
    });
    document.getElementById('reset-btn').addEventListener('click', initGame);
    document.getElementById('undo-btn').addEventListener('click', undo);

    initGame();
</script>
